
/* ---- RBC in 3D: the band-mounted rig, interactive --------------------------
   Painter's-algorithm renderer over prisms and boxes: plain 2D canvas, no WebGL,
   no library, so the page stays one self-contained file. Layout settled in
   scene_v3.py and transcribed here -- keep the constants in step with that file.

   Shaft along Y. Two end plates 6 ft apart, wafer pack between, two steel bands at
   0 and 180 deg running plate to plate at r/R = 0.95, four divots per band, nine
   coupons per divot. Lamps sit in six holes in the cover, three at each end, so
   PHOTOPERIOD IS AXIAL: one end summer, the other winter.

   INTERACTION
     - drag            orbit
     - wheel-angle     rotate the shaft; both bands and all divots follow
     - DAY slider      recolors every coupon by its predicted biomass on that day
     - click a coupon  reads out that coupon's predicted state on the selected day

   Coupon color is brightness-scaled by X(day) relative to the brightest divot on
   the same day, so the growth you see is relative, not absolute -- the readout
   carries the numbers.
--------------------------------------------------------------------------- */
(function(){
  var cv=document.getElementById('scene'); if(!cv) return;
  var ctx=cv.getContext('2d');
  var R=1.00, HS=0.9144, PLATE_T=0.030, WAF_R=0.90, N_WAF=11;
  var BAND_R=0.95, BAND_W=0.125, BAND_T=0.028;
  var DIVOT_Y=[-0.70,-0.42,0.42,0.70];
  var COUP=0.0397, COUP_VIZ=1.80;
  var WATER=-0.1577*R, TX=1.24, TY=1.08, TZ0=-1.06, ZCOV=1.30;
  var LAMP_Y=[-0.72,0.72], LAMP_X=[-0.40,0.0,0.40];
  var A_EXP_M2=0.00129008;          /* one face net of one magnet */
  var MATCOL={PE:[95,179,224],STL:[185,192,200],GRA:[192,140,240],FAB:[79,201,160]};
  var MATFULL={PE:'HDPE',STL:'bare steel',GRA:'graphene-coated steel',FAB:'non-woven fabric'};
  var BANDS=[{phi:0,       mats:['PE','STL','STL','PE']},
             {phi:Math.PI, mats:['FAB','GRA','GRA','FAB']}];
  var view={yaw:-1.18,pitch:0.34,dist:4.60}, drag=null, moved=0, phase=0.95, day=60, sel=null;

  function val(id,dflt){var e=document.getElementById(id);return e?e.value:dflt;}
  /* key order follows build_grid_v3: material|photoperiod|rpm|n_shear|shelter|load */
  function keyFor(mat,end){
    return [mat,end,val('rpm','4.0'),val('nsh','0.0'),val('shel','1.0'),val('load','low')].join('|');
  }
  function at(mat,end,d){
    var s=(typeof SERIES!=='undefined')?SERIES[keyFor(mat,end)]:null;
    if(!s) return null;
    var i=Math.max(0,Math.min(s.t.length-1,Math.round(d)));
    var Xp=s.Xp[i],Xn=s.Xn[i],Xh=s.Xh[i];
    return {Xp:Xp,Xn:Xn,Xh:Xh,X:Xp+Xn+Xh,Lf:s.Lf[i],JA:s.JA[i],t:s.t[i]};
  }
  function maxX(d){
    var m=0;
    BANDS.forEach(function(b){ b.mats.forEach(function(mt,i){
      var o=at(mt, DIVOT_Y[i]<0?'winter':'summer', d); if(o) m=Math.max(m,o.X); });});
    return m||1;
  }

  function rot(p){
    var cy=Math.cos(view.yaw), sy=Math.sin(view.yaw),
        cp=Math.cos(view.pitch), sp=Math.sin(view.pitch);
    var x=p[0]*cy-p[1]*sy, y=p[0]*sy+p[1]*cy, z=p[2];
    return [x, y*cp-z*sp, y*sp+z*cp];
  }
  function proj(p,w,h){ var q=rot(p), f=h*0.92, s=f/(view.dist+q[1]);
    return [w/2+q[0]*s, h/2-q[2]*s, view.dist+q[1]]; }
  function shade(col,nrm){
    var L=[0.42,-0.72,0.55], n=Math.hypot(nrm[0],nrm[1],nrm[2])||1;
    var d=(nrm[0]*L[0]+nrm[1]*L[1]+nrm[2]*L[2])/n, k=0.56+0.44*Math.max(0,d);
    return 'rgb('+col.map(function(c){return Math.round(Math.min(255,c*k));}).join(',')+')';
  }
  var faces=[];
  /* `order` pins a face past the depth sort: a large translucent quad always wins on
     centroid depth, which flipped the water plane and the cover in front of the wheel. */
  function quad(p,col,a,nrm,order,meta){
    faces.push({p:p,c:col,a:(a===undefined?1:a),s:nrm||[0,0,1],o:order||0,m:meta||null}); }
  function boxq(c,s,col,a,order){
    var X=[c[0]-s[0],c[0]+s[0]],Y=[c[1]-s[1],c[1]+s[1]],Z=[c[2]-s[2],c[2]+s[2]];
    quad([[X[0],Y[0],Z[1]],[X[1],Y[0],Z[1]],[X[1],Y[1],Z[1]],[X[0],Y[1],Z[1]]],col,a,[0,0,1],order);
    quad([[X[0],Y[0],Z[0]],[X[1],Y[0],Z[0]],[X[1],Y[1],Z[0]],[X[0],Y[1],Z[0]]],col,a,[0,0,-1],order);
    quad([[X[0],Y[0],Z[0]],[X[0],Y[0],Z[1]],[X[0],Y[1],Z[1]],[X[0],Y[1],Z[0]]],col,a,[-1,0,0],order);
    quad([[X[1],Y[0],Z[0]],[X[1],Y[0],Z[1]],[X[1],Y[1],Z[1]],[X[1],Y[1],Z[0]]],col,a,[1,0,0],order);
    quad([[X[0],Y[0],Z[0]],[X[1],Y[0],Z[0]],[X[1],Y[0],Z[1]],[X[0],Y[0],Z[1]]],col,a,[0,-1,0],order);
    quad([[X[0],Y[1],Z[0]],[X[1],Y[1],Z[0]],[X[1],Y[1],Z[1]],[X[0],Y[1],Z[1]]],col,a,[0,1,0],order);
  }
  function prismY(y0,y1,rad,n,col,capcol,a,order){
    var i,a0,a1;
    for(i=0;i<n;i++){
      a0=2*Math.PI*i/n; a1=2*Math.PI*(i+1)/n;
      quad([[rad*Math.cos(a0),y0,rad*Math.sin(a0)],[rad*Math.cos(a1),y0,rad*Math.sin(a1)],
            [rad*Math.cos(a1),y1,rad*Math.sin(a1)],[rad*Math.cos(a0),y1,rad*Math.sin(a0)]],
           col,a,[Math.cos((a0+a1)/2),0,Math.sin((a0+a1)/2)],order);
    }
    for(var s=0;s<2;s++){
      var y=s?y1:y0, ring=[];
      for(i=0;i<n;i++){ var t=2*Math.PI*i/n; ring.push([rad*Math.cos(t),y,rad*Math.sin(t)]); }
      quad(ring, capcol||col, a, [0,s?1:-1,0], order);
    }
  }
  /* band-local frame: u = radial offset from BAND_R, v = tangential, y = axial */
  function atb(phi,u,v,y){ var ca=Math.cos(phi), sa=Math.sin(phi), r=BAND_R+u;
    return [r*ca-v*sa, y, r*sa+v*ca]; }
  function bandPrism(phi,y0,y1,u0,u1,v0,v1,col,a,order,meta){
    var ca=Math.cos(phi), sa=Math.sin(phi);
    var P=function(u,v,y){return atb(phi,u,v,y);};
    quad([P(u1,v0,y0),P(u1,v1,y0),P(u1,v1,y1),P(u1,v0,y1)],col,a,[ ca,0, sa],order,meta);
    quad([P(u0,v0,y0),P(u0,v1,y0),P(u0,v1,y1),P(u0,v0,y1)],col,a,[-ca,0,-sa],order,meta);
    quad([P(u0,v1,y0),P(u1,v1,y0),P(u1,v1,y1),P(u0,v1,y1)],col,a,[-sa,0, ca],order,meta);
    quad([P(u0,v0,y0),P(u1,v0,y0),P(u1,v0,y1),P(u0,v0,y1)],col,a,[ sa,0,-ca],order,meta);
    quad([P(u0,v0,y1),P(u1,v0,y1),P(u1,v1,y1),P(u0,v1,y1)],col,a,[0,1,0],order,meta);
    quad([P(u0,v0,y0),P(u1,v0,y0),P(u1,v1,y0),P(u0,v1,y0)],col,a,[0,-1,0],order,meta);
  }
  function litEnds(){
    /* the control is id="phot", not "photo" -- with the wrong id this silently fell
       through to 'both' and the lamps never responded to the photoperiod selector */
    var v=val('phot','__both');
    if(v==='__both') return {summer:1,winter:1};
    return {summer:(v==='summer')?1:0, winter:(v==='winter')?1:0};
  }
  function build(){
    faces=[];
    var lit=litEnds(), xm=maxX(day);
    quad([[-TX,-TY,TZ0],[TX,-TY,TZ0],[TX,TY,TZ0],[-TX,TY,TZ0]],[38,43,49],1,[0,0,1],0);
    prismY(-TY-0.05,TY+0.05,0.045,18,[168,172,178],[150,154,160],1,0);
    for(var i=0;i<N_WAF;i++){
      var y=-HS+0.16+(2*HS-0.32)*i/(N_WAF-1);
      prismY(y-0.009,y+0.009,WAF_R,48,[96,102,109],[86,92,99],1,0);
    }
    /* End plates translucent FOR LEGIBILITY ONLY -- solid steel in reality, but opaque
       they hide the wafer pack, both bands and every divot behind them. Order 0 so they
       depth-sort: pinned later they drew OVER the far-end divots and washed them out. */
    [-1,1].forEach(function(sg){
      prismY(sg*HS-PLATE_T,sg*HS+PLATE_T,R,56,[120,126,133],[104,110,117],0.42,0);
    });
    BANDS.forEach(function(bd,bi){
      var phi=bd.phi+phase;
      /* The band is drawn in SEGMENTS. A painter's algorithm sorts by centroid depth,
         so one prism spanning the whole shaft got a single depth and repainted over
         every divot on its far half. */
      var NSEG=16;
      for(var q=0;q<NSEG;q++){
        var y0=-HS+2*HS*q/NSEG, y1=-HS+2*HS*(q+1)/NSEG;
        bandPrism(phi,y0,y1,-BAND_T,BAND_T,-BAND_W,BAND_W,[168,175,184],1,0);
      }
      DIVOT_Y.forEach(function(y,di){
        var dy=3*COUP*COUP_VIZ*0.5+0.014;
        bandPrism(phi,y-dy,y+dy,BAND_T-0.010,BAND_T+0.002,-dy,dy,[52,57,63],1,0);
        var end=(y<0)?'winter':'summer', on=lit[end], mat=bd.mats[di];
        var st=at(mat,end,day);
        /* brightness tracks predicted biomass on the selected day, floored so an
           early-day coupon is still visible rather than black */
        var g=st?(0.30+0.70*Math.min(1,st.X/xm)):0.75;
        if(!on) g*=0.45;
        var base=MATCOL[mat];
        for(var a=0;a<3;a++) for(var b=0;b<3;b++){
          var ci=a*3+b;
          var isSel=sel&&sel.bi===bi&&sel.di===di&&sel.ci===ci;
          var gg=isSel?Math.min(1,g+0.40):g;
          var col=[Math.round(base[0]*gg),Math.round(base[1]*gg),Math.round(base[2]*gg)];
          var cs=COUP*COUP_VIZ, cy=y+(a-1)*cs*1.06, cvv=(b-1)*cs*1.06;
          bandPrism(phi,cy-cs/2,cy+cs/2,BAND_T-0.004,BAND_T+(isSel?0.016:0.008),
                    cvv-cs/2,cvv+cs/2,col,1,0,{bi:bi,di:di,ci:ci,mat:mat,end:end});
        }
      });
    });
    quad([[-TX,-TY,WATER],[TX,-TY,WATER],[TX,TY,WATER],[-TX,TY,WATER]],[26,92,86],0.55,[0,0,1],2);
    quad([[-TX,-TY,ZCOV],[TX,-TY,ZCOV],[TX,TY,ZCOV],[-TX,TY,ZCOV]],[78,86,95],0.26,[0,0,1],3);
    LAMP_Y.forEach(function(ly){
      var end=(ly<0)?'winter':'summer', on=lit[end];
      var body=(end==='summer')?[214,158,70]:[92,144,214];
      if(!on) body=[Math.round(body[0]*0.36),Math.round(body[1]*0.36),Math.round(body[2]*0.36)];
      LAMP_X.forEach(function(lx){
        boxq([lx,ly,ZCOV-0.02],[0.080,0.080,0.028],[70,76,83],1,4);
        boxq([lx,ly,ZCOV-0.085],[0.066,0.066,0.042],body,1,4);
        boxq([lx,ly,ZCOV-0.135],[0.055,0.055,0.010],
             [Math.min(255,body[0]+45),Math.min(255,body[1]+45),Math.min(255,body[2]+45)],1,4);
      });
    });
  }
  var LASTW=0, LASTH=0;
  function draw(){
    var w=cv.parentNode.clientWidth-26, h=Math.round(w*0.58), DPR=Math.max(1,devicePixelRatio||1);
    if(w<=0) return;        /* the tab is hidden at parse time; redraw happens on switch */
    LASTW=w; LASTH=h;
    cv.width=w*DPR; cv.height=h*DPR; cv.style.height=h+'px';
    ctx.setTransform(DPR,0,0,DPR,0,0); ctx.clearRect(0,0,w,h);
    build();
    faces.forEach(function(f){ var d=0;
      f.q=f.p.map(function(p){var r=proj(p,w,h); d+=r[2]; return r;}); f.d=d/f.p.length; });
    faces.sort(function(a,b){ return (a.o-b.o) || (b.d-a.d); });
    faces.forEach(function(f){
      ctx.beginPath();
      f.q.forEach(function(p,i){ i?ctx.lineTo(p[0],p[1]):ctx.moveTo(p[0],p[1]); });
      ctx.closePath();
      ctx.globalAlpha=f.a; ctx.fillStyle=shade(f.c,f.s); ctx.fill();
      ctx.globalAlpha=Math.min(1,f.a+0.25);
      var isSel=f.m&&sel&&f.m.bi===sel.bi&&f.m.di===sel.di&&f.m.ci===sel.ci;
      ctx.strokeStyle=isSel?'#ffffff':'rgba(12,14,16,.55)';
      ctx.lineWidth=isSel?1.6:0.8; ctx.stroke(); ctx.globalAlpha=1;
    });
  }
  function inside(pt,poly){
    var x=pt[0],y=pt[1],c=false;
    for(var i=0,j=poly.length-1;i<poly.length;j=i++){
      var xi=poly[i][0],yi=poly[i][1],xj=poly[j][0],yj=poly[j][1];
      if(((yi>y)!==(yj>y)) && (x < (xj-xi)*(y-yi)/(yj-yi)+xi)) c=!c;
    }
    return c;
  }
  function pick(px,py){
    /* Faces are drawn far-to-near, so walk the list backwards for front-most first.
       TRANSLUCENT faces are skipped: the water plane and the cover are pinned to draw
       last and span most of the canvas, so without this they intercept every click.
       Measured on the Python replica of this same projection: at a wheel angle of 0 or
       180 degrees only 2 of 72 coupons were clickable without the filter, against 22
       with it. The translucent end plates are skipped for the same reason -- a coupon
       seen through one should still be selectable. */
    for(var i=faces.length-1;i>=0;i--){
      var f=faces[i];
      if(f.a<0.9) continue;
      if(f.q && inside([px,py], f.q)){ return f.m||null; }
    }
    return null;
  }
  function fmt(v,n){ return (v===undefined||v===null||!isFinite(v))?'\u2014':v.toFixed(n); }
  function readout(){
    var el=document.getElementById('cpinfo'); if(!el) return;
    if(!sel){ el.innerHTML='<div class="rdhint">Click any coupon to read its predicted '+
      'state on the selected day.</div>'; return; }
    var st=at(sel.mat,sel.end,day);
    if(!st){ el.innerHTML='<div class="rdhint">That arm was not run.</div>'; return; }
    var tot=st.X||1e-9;
    var pct=function(v){return (100*v/tot).toFixed(0);};
    var mg1=st.X*A_EXP_M2*1000, mg3=mg1*3;
    var harvest = (Math.abs(day-5)<0.5||Math.abs(day-7)<0.5||Math.abs(day-10)<0.5)
      ? '<span class="rdtag">harvest day</span>' : '';
    el.innerHTML =
      '<div class="rdhead"><span class="sw" style="background:rgb('+MATCOL[sel.mat].join(',')+')"></span>'+
      '<strong>'+MATFULL[sel.mat]+'</strong> &nbsp;&middot;&nbsp; '+sel.end+' end'+
      ' &nbsp;&middot;&nbsp; divot '+(sel.di+1)+', coupon '+(sel.ci+1)+' of 9'+
      ' &nbsp;&middot;&nbsp; day '+day+' '+harvest+'</div>'+
      '<div class="rdbar">'+
        '<span style="width:'+pct(st.Xp)+'%;background:#4fc3a1"></span>'+
        '<span style="width:'+pct(st.Xn)+'%;background:#5a9fe0"></span>'+
        '<span style="width:'+pct(st.Xh)+'%;background:#e8835a"></span></div>'+
      '<div class="rdgrid">'+
        '<div><span class="k">total biofilm</span><span class="v">'+fmt(st.X,2)+'</span>'+
          '<span class="u">g DW m\u207B\u00B2</span></div>'+
        '<div><span class="k">phototrophs</span><span class="v" style="color:#4fc3a1">'+
          fmt(st.Xp,2)+'</span><span class="u">'+pct(st.Xp)+' % of film</span></div>'+
        '<div><span class="k">nitrifiers</span><span class="v" style="color:#5a9fe0">'+
          fmt(st.Xn,3)+'</span><span class="u">'+pct(st.Xn)+' % of film</span></div>'+
        '<div><span class="k">heterotrophs</span><span class="v" style="color:#e8835a">'+
          fmt(st.Xh,2)+'</span><span class="u">'+pct(st.Xh)+' % of film</span></div>'+
        '<div><span class="k">film thickness</span><span class="v">'+fmt(st.Lf*1000,1)+
          '</span><span class="u">\u00B5m</span></div>'+
        '<div><span class="k">removal flux</span><span class="v">'+fmt(st.JA,3)+
          '</span><span class="u">g N m\u207B\u00B2 d\u207B\u00B9</span></div>'+
        '<div><span class="k">this coupon</span><span class="v">'+fmt(mg1,1)+
          '</span><span class="u">mg DW on A_exp</span></div>'+
        '<div><span class="k">3-coupon harvest</span><span class="v">'+fmt(mg3,1)+
          '</span><span class="u">mg DW, lower bound</span></div>'+
      '</div>';
  }
  function refresh(){ draw(); readout(); }

  cv.style.cursor='pointer';
  cv.addEventListener('pointerdown',function(e){drag=[e.clientX,e.clientY];moved=0;
    cv.setPointerCapture(e.pointerId);});
  cv.addEventListener('pointerup',function(e){
    if(drag && moved<5){                       /* a click, not an orbit */
      var r=cv.getBoundingClientRect();
      var m=pick(e.clientX-r.left, e.clientY-r.top);
      sel = m ? m : null;
      refresh();
    }
    drag=null;});
  cv.addEventListener('pointermove',function(e){
    if(!drag) return;
    moved += Math.abs(e.clientX-drag[0])+Math.abs(e.clientY-drag[1]);
    view.yaw += (e.clientX-drag[0])*0.011;
    view.pitch = Math.max(-0.25, Math.min(1.15, view.pitch + (e.clientY-drag[1])*0.008));
    drag=[e.clientX,e.clientY]; draw();
  });
  var ph=document.getElementById('scphase');
  if(ph) ph.addEventListener('input',function(){
    phase=parseFloat(ph.value)*Math.PI/180;
    var lab=document.getElementById('scphaselab');
    if(lab) lab.textContent=ph.value+String.fromCharCode(176);
    draw();
  });
  var dy=document.getElementById('scday');
  if(dy) dy.addEventListener('input',function(){
    day=parseInt(dy.value,10);
    var lab=document.getElementById('scdaylab');
    if(lab) lab.textContent='day '+day;
    refresh();
  });
  ['phot','rpm','nsh','shel','load','mat'].forEach(function(id){
    var e=document.getElementById(id); if(e) e.addEventListener('change',refresh);
  });
  window.addEventListener('resize',refresh);
  window.SCENE_DRAW=refresh;
  refresh();
})();
